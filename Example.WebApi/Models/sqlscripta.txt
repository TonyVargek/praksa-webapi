CREATE TABLE "Store" (
    Id UUID PRIMARY KEY,
    Name VARCHAR(255),
    Type VARCHAR(100),
    Owner VARCHAR(255),
    Address VARCHAR(500)
);

CREATE TABLE "User" (
    Id UUID PRIMARY KEY,
    FirstName VARCHAR(255),
    LastName VARCHAR(255),
    CreatedAt TIMESTAMP,
    EmailAddress VARCHAR(255),
    PhoneNumber VARCHAR(50)
);

CREATE TABLE "DeliveryPerson" (
    Id UUID PRIMARY KEY,
    FirstName VARCHAR(255),
    LastName VARCHAR(255),
    Vehicle VARCHAR(100),
    Age INTEGER,
    Fulltime BOOLEAN
);

CREATE TABLE "Product" (
    Id UUID PRIMARY KEY,
    Name VARCHAR(255),
    Price NUMERIC(10,2),
    Type VARCHAR(100),
    Description TEXT,
    StoreId UUID not null references "Store"(Id),
);

CREATE TABLE "Order" (
    Id UUID PRIMARY KEY,
    OrderTime TIMESTAMP,
    Total NUMERIC(10,2),
    UserId UUID references "User"(Id) NOT NULL,
    DeliveryPersonId UUID references "DeliveryPerson"(Id),
    Status VARCHAR(50)
);

CREATE TABLE "OrderProduct" (
    Id UUID PRIMARY KEY,
    OrderId UUID references "Order"(Id) NOT NULL,
    ProductId UUID references "Product"(Id) NOT NULL,
    Quantity INTEGER NOT NULL
);

select u.FirstName, u.LastName, dp.FirstName, dp.LastName 
	from "Order" as o inner join "User" as u on u.Id = o.UserId 
	right join "DeliveryPerson" as dp on dp.Id = o.DeliveryPersonId;

select p.Name, p.Price,  s.Name, s.Address 
	from "Product" as p inner join "Store" as s on p.StoreId = s.Id 
	where p.Price > 7.0

select u.FirstName, u.LastName, o.Id as "Broj narudzbe", SUM(op.Quantity) as "Broj proizvoda" 
	from "User" as u inner join "Order" as o on u.Id = o.UserId 
	inner join "OrderProduct" as op on op.OrderId = o.Id 
	group by u.FirstName, u.LastName, o.Id
	order by "Broj proizvoda" ASC

select s.Name, s.Owner, s.Address, SUM(op.Quantity) as "Ukupan broj narucenih proizvoda" 
	from "Store" as s inner join "Product" as p on s.Id = p.StoreId 
	inner join "OrderProduct" as op on op.ProductId = p.Id 
	group by s.Name, s.Owner, s.Address
	order by "Ukupan broj narucenih proizvoda" DESC

select u.FirstName, u.LastName, o.Id as "Broj narudzbe", count(distinct s.Id) as "Broj razlicitih trgovina" 
	from "User" as u inner join "Order" as o on u.Id = o.UserId 
	inner join "OrderProduct" as op on o.Id = op.OrderId 
	inner join "Product" as p on op.ProductId = p.Id
	inner join "Store" as s on s.Id = p.StoreId
	group by u.FirstName, u.LastName, o.Id
	having count(distinct s.Id) > 1
